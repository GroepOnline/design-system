<script>
const start = performance.now();
const steps = [];
function fmt(ms){const s=Math.floor(ms/1000);return `${Math.floor(s/60)}m ${String(s%60).padStart(2,"0")}s`}
(function tick(){document.getElementById("t1").textContent=fmt(performance.now()-start);requestAnimationFrame(tick)})();
const icons={r:"#i-file",w:"#i-pencil",x:"#i-terminal"};
const labels={r:"lees src/styles/*.scss",w:"schrijf _text-gradient.scss",x:"curl productie-domein"};
function rip(kind){
  const box=document.getElementById("tools1");
  const el=document.createElement("span");
  el.className=`ripple ${kind} on`;box.appendChild(el);
  steps.push({kind,t:fmt(performance.now()-start)});
  document.getElementById("d1").innerHTML=steps.map(s=>
    `<div class="step"><span class="st">${s.t}</span><svg class="ic"><use href="${icons[s.kind]}"/></svg><span>${labels[s.kind]}</span></div>`).join("");
  setTimeout(()=>{el.classList.remove("on");setTimeout(()=>el.remove(),400)},2600);
}
</script>
