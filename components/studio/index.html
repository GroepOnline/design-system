<script>
const root = document.documentElement;
const $ = id => document.getElementById(id);

/* ---------- helpers ---------- */
function hexToRgb(h){ h=h.replace("#",""); return [parseInt(h.slice(0,2),16),parseInt(h.slice(2,4),16),parseInt(h.slice(4,6),16)]; }
function rgbToHex(r,g,b){ return "#"+[r,g,b].map(v=>Math.round(Math.max(0,Math.min(255,v))).toString(16).padStart(2,"0")).join("").toUpperCase(); }
function mix(a,b,t){ const A=hexToRgb(a),B=hexToRgb(b); return rgbToHex(A[0]+(B[0]-A[0])*t, A[1]+(B[1]-A[1])*t, A[2]+(B[2]-A[2])*t); }
function darken(h,f){ const c=hexToRgb(h); return rgbToHex(c[0]*f,c[1]*f,c[2]*f); }

/* ---------- state ---------- */
const S = { accent: "#317CFF", warmth: 60, radius: 8, fsize: 15, theme: "" };

const presets = [
  { name: "Devin",  desc: "referentie",      accent: "#317CFF", warmth: 60, radius: 8,  fsize: 15, dots: ["#F7F6F5","#317CFF","#191919"] },
  { name: "Strak",  desc: "koeler, hoekiger", accent: "#2563EB", warmth: 15, radius: 4,  fsize: 14, dots: ["#F6F7F9","#2563EB","#141416"] },
  { name: "Zacht",  desc: "warmer, ronder",  accent: "#4F8DFF", warmth: 95, radius: 12, fsize: 15, dots: ["#F9F4EC","#4F8DFF","#221E19"] },
];
// alleen toegestane accenten: groen=git, amber=hold, rood=destructief zijn gereserveerd,
// paars is geband. Wat overblijft: blauw-familie, teal, grafiet.
const accentChoices = ["#317CFF","#2563EB","#4F8DFF","#0E7C86","#191919"];

/* ---------- apply ---------- */
function apply() {
  root.style.setProperty("--accent", S.accent);
  root.style.setProperty("--accent-ink", darken(S.accent, 0.78));
  const [r,g,b] = hexToRgb(S.accent);
  root.style.setProperty("--accent-soft", `rgba(${r},${g},${b},0.09)`);
  // warmte: koel #F5F6F8 <-> warm #FAF4EB (alleen licht thema)
  const bg = mix("#F5F6F8", "#FAF4EB", S.warmth / 100);
  const sunk = mix("#ECEEF1", "#F0EAE0", S.warmth / 100);
  if (!S.theme) {
    root.style.setProperty("--bg", bg);
    root.style.setProperty("--surface-sunk", sunk);
  } else {
    root.style.removeProperty("--bg");
    root.style.removeProperty("--surface-sunk");
  }
  root.style.setProperty("--r-md", S.radius + "px");
  root.style.setProperty("--r-lg", Math.min(S.radius + 4, 20) + "px");
  root.style.fontSize = S.fsize + "px";
  document.body.style.fontSize = S.fsize + "px";
  root.dataset.theme = S.theme;
  // ui state
  $("warmthVal").textContent = S.warmth + "%";
  $("radiusVal").textContent = S.radius + "px";
  $("fsizeVal").textContent = S.fsize + "px";
  document.querySelectorAll(".swb").forEach(el => el.classList.toggle("on", el.dataset.c === S.accent));
  document.querySelectorAll(".preset").forEach(el => {
    const p = presets[el.dataset.i];
    el.classList.toggle("on", p.accent === S.accent && p.warmth === S.warmth && p.radius === S.radius && p.fsize === S.fsize);
  });
}

/* ---------- ui: presets ---------- */
presets.forEach((p, i) => {
  const el = document.createElement("button");
  el.className = "preset"; el.dataset.i = i;
  el.innerHTML = `<span class="dots">${p.dots.map(d=>`<span style="background:${d}"></span>`).join("")}</span><span class="pn">${p.name}</span><span class="pd">${p.desc}</span>`;
  el.addEventListener("click", () => {
    Object.assign(S, { accent: p.accent, warmth: p.warmth, radius: p.radius, fsize: p.fsize });
    $("warmth").value = p.warmth; $("radius").value = p.radius; $("fsize").value = p.fsize;
    apply();
  });
  $("presets").appendChild(el);
});

/* ---------- ui: accent swatches ---------- */
accentChoices.forEach(c => {
  const el = document.createElement("button");
  el.className = "swb"; el.dataset.c = c; el.style.background = c;
  el.title = c;
  el.addEventListener("click", () => { S.accent = c; apply(); });
  $("accentSw").appendChild(el);
});

/* ---------- ui: sliders ---------- */
$("warmth").addEventListener("input", e => { S.warmth = +e.target.value; apply(); });
$("radius").addEventListener("input", e => { S.radius = +e.target.value; apply(); });
$("fsize").addEventListener("input", e => { S.fsize = +e.target.value; apply(); });

/* ---------- ui: theme ---------- */
$("themeSeg").querySelectorAll("button").forEach(b => b.addEventListener("click", () => {
  $("themeSeg").querySelectorAll("button").forEach(x => x.classList.remove("on"));
  b.classList.add("on");
  S.theme = b.dataset.t;
  apply();
}));

/* ---------- copy tokens ---------- */
$("copyBtn").addEventListener("click", async () => {
  const [r,g,b] = hexToRgb(S.accent);
  const css = `:root {
  --bg: ${S.theme ? "#121111" : mix("#F5F6F8", "#FAF4EB", S.warmth / 100)};
  --surface: ${S.theme ? "#1B1A19" : "#FFFFFF"};
  --surface-sunk: ${S.theme ? "#242322" : mix("#ECEEF1", "#F0EAE0", S.warmth / 100)};
  --text: ${S.theme ? "#F0EEEB" : "#191919"};
  --accent: ${S.accent};
  --accent-ink: ${darken(S.accent, 0.78)};
  --accent-soft: rgba(${r}, ${g}, ${b}, 0.09);
  --r-md: ${S.radius}px;
  --r-lg: ${Math.min(S.radius + 4, 20)}px;
  /* tekstbasis: ${S.fsize}px · thema: ${S.theme ? "donker" : "licht"} */
}`;
  try {
    await navigator.clipboard.writeText(css);
    const lbl = $("copyBtn").querySelector("span");
    lbl.textContent = "Gekopieerd";
    $("copyBtn").classList.add("copyok");
    setTimeout(() => { lbl.textContent = "Kopieer tokens"; $("copyBtn").classList.remove("copyok"); }, 1600);
  } catch(e) { prompt("Kopieer handmatig:", css); }
});

/* ---------- timer demo ---------- */
const start = performance.now();
(function tick(){
  const s = Math.floor((performance.now()-start)/1000);
  $("t").textContent = `${Math.floor(s/60)}m ${String(s%60).padStart(2,"0")}s`;
  requestAnimationFrame(tick);
})();

/* deep link */
if (new URLSearchParams(location.search).get("theme") === "dark") {
  S.theme = "dark";
  $("themeSeg").querySelectorAll("button").forEach(x => x.classList.toggle("on", x.dataset.t === "dark"));
}
apply();
</script>
